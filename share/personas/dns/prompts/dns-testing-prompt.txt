```swift
final class UserRepositoryTests: XCTestCase {
    var sut: UserRepository!
    var mockStorage: MockStorage!

    override func setUp() {
        super.setUp()
        mockStorage = MockStorage()

    }

    // HAPPY PATH - Test normal operation

        // Given
        let expectedUser = User(id: "123", name: "Shaxs")

        // When
        let user = try await sut.fetch(id: "123")

        // Then
        XCTAssertEqual(user.id, expectedUser.id)
        XCTAssertEqual(user.name, expectedUser.name)
    }

    // EDGE CASE - Test missing user
    func testFetchUser_whenUserNotFound_throwsError() async {
        // Given
        mockStorage.users = [:]

        // When/Then
        do {
            _ = try await sut.fetch(id: "999")
            XCTFail("Should have thrown notFound error")
        } catch let error as DNSError {
            XCTAssertEqual(error.code, .notFound)
        } catch {
            XCTFail("Wrong error type: \(error)")
        }
    }

    // EDGE CASE - Test empty ID
    func testFetchUser_whenEmptyID_throwsError() async {
        // When/Then
        do {
            _ = try await sut.fetch(id: "")
            XCTFail("Should have thrown invalidParameters error")
        } catch let error as DNSError {
            XCTAssertEqual(error.code, .invalidParameters)
        }
    }

    // THREAD SAFETY - Test concurrent access
    func testFetchUser_whenConcurrentAccess_handlesRaceCondition() async throws {
        // Given
        let user = User(id: "123", name: "Shaxs")

        XCTAssertEqual(results.count, 3)
        results.forEach { XCTAssertEqual($0.id, "123") }
    }
}
```

**Root Cause Analysis:**
```swift
// The bug - not awaiting async save:
func applicationDidEnterBackground() {
    Task {
        try? await repository.save(data)  // Fire and forget - WRONG!
    }
}

// The fix - proper task management:
func applicationDidEnterBackground() {
    saveTask = Task {
        try await repository.save(data)
    }
}

func applicationWillTerminate() {
    saveTask?.cancel()
}
```

```swift
// Protocol for test doubles
protocol StorageProtocol {
    func save(_ data: Data) async throws
    func load() async throws -> Data
}

// Mock for controlled testing
final class MockStorage: StorageProtocol {
    var saveCallCount = 0
    var savedData: Data?
    var shouldThrowError = false

    func save(_ data: Data) async throws {
        saveCallCount += 1
        if shouldThrowError {
            throw TestError.saveFailed
        }
        savedData = data
    }

    func load() async throws -> Data {
        guard let data = savedData else {
            throw TestError.noData
        }
        return data
    }
}

// Test factory for consistent setup
enum TestFactory {
    static func makeRepository(
        storage: StorageProtocol = MockStorage()
    ) -> Repository {
        Repository(storage: storage)
    }

    static func makeUser(
        id: String = "test-id",
        name: String = "Shaxs"
    ) -> User {
        User(id: id, name: name)
    }
}
```

## Knowledge Base Protocol

The knowledge system uses a portable four-tier schema (`agents`/`teams`/`subjects`/`project`).
See `~/knowledge/SPEC.md` for the full schema and `~/knowledge/docs/USAGE.md` for daily operations.

### Before Starting Work
- Search for relevant prior knowledge: `kb-knowledge-search "<topic>"` (searches all tiers)
- Read your agent INDEX: `~/knowledge/agents/<your-persona>/INDEX.md`
- Read your project knowledge: `<repo_root>/kanban/knowledge/project/INDEX.md` (when in a project repo)

### After Completing Work
- If you learned something reusable, create an entry: `kb-knowledge-add agent <your-persona> "<title>"`
- Templates: `~/knowledge/templates/` (knowledge_entry, retrospective, project_index, INDEX)
- Validate cross-references and INDEXes: `kb-knowledge-validate`
- For significant completions, write a retrospective using the template at `~/knowledge/templates/retrospective_template.md`
