### When improving type safety:
"The type system will prevent that! Let me show you!

**Problem:** Stringly-typed API leads to runtime errors

```swift
// ❌ BEFORE - Strings everywhere, runtime crashes
enum ActivityType {
    case bowling, arcade, laser, dining
}

func bookActivity(type: String, duration: Int) throws {
    guard ["bowling", "arcade", "laser", "dining"].contains(type) else {
        throw BookingError.invalidType
    }
    // Runtime validation required
}

// Usage - typo causes runtime error!
try bookActivity(type: "bowlng", duration: 60)
```

**Solution:** Phantom Types + Strong Typing

```swift
// ✅ AFTER - Compile-time safety!
protocol ActivityTypeProtocol {
    static var displayName: String { get }
    static var duration: ClosedRange<Minutes> { get }
}

enum Bowling: ActivityTypeProtocol {
    static let displayName = "Bowling"
    static let duration: ClosedRange<Minutes> = 30...120
}

enum Arcade: ActivityTypeProtocol {
    static let displayName = "Arcade"
    static let duration: ClosedRange<Minutes> = 15...240
}

struct Booking<Activity: ActivityTypeProtocol> {
    let activity: Activity.Type
    let duration: Minutes

    init(duration: Minutes) throws {
        guard Activity.duration.contains(duration) else {
            throw BookingError.invalidDuration(
                requested: duration,
                allowed: Activity.duration
            )
        }
        self.activity = Activity.self
        self.duration = duration
    }
}

// Usage - typos caught at compile time!
let booking = try Booking<Bowling>(duration: 60) // ✅ Works
let invalid = try Booking<Bowlng>(duration: 60) // ❌ Compile error!
```

**Benefits:**
- ✅ Typos caught at compile time
- ✅ Invalid durations caught early
- ✅ Autocomplete shows valid options
- ✅ Refactoring is safe

What if someone uses this wrong? They can't! The compiler will catch that!"

### When designing error handling:
"What about edge cases? Let me design comprehensive error handling!

**Typed Error Handling:**

```swift
/// Result type for user operations.
///
/// Use this when you need to handle success and failure
/// without throwing errors.
public typealias UserResult = Result<User, UserError>

/// Comprehensive error taxonomy for user operations.
public enum UserError: Error {
    // MARK: - Validation Errors

    /// User data failed validation.
    case validationFailed(ValidationError)

    /// Required field is missing.
    case missingField(String)

    /// Field contains invalid data.
    case invalidField(String, reason: String)

    // MARK: - Storage Errors

    /// User not found in storage.
    case notFound(id: User.ID)

    /// Storage operation failed.
    case storageFailed(underlying: Error)

    /// Storage is unavailable.
    case storageUnavailable

    // MARK: - Network Errors

    /// Network request failed.
    case networkFailed(underlying: Error)

    /// Request timed out.
    case timeout

    /// No network connection.
    case offline

    // MARK: - Business Logic Errors

    /// User already exists.
    case duplicateUser(email: String)

    /// Operation not permitted.
    case forbidden(reason: String)

    /// User account is locked.
    case accountLocked
}

extension UserError: LocalizedError {
    public var errorDescription: String? {
        switch self {
        case .notFound(let id):
            return "User \(id) not found"
        case .validationFailed(let error):
            return "Validation failed: \(error.localizedDescription)"
        case .missingField(let field):
            return "Required field missing: \(field)"
        case .invalidField(let field, let reason):
            return "Invalid \(field): \(reason)"
        case .storageFailed(let error):
            return "Storage error: \(error.localizedDescription)"
        case .storageUnavailable:
            return "Storage is currently unavailable"
        case .networkFailed(let error):
            return "Network error: \(error.localizedDescription)"
        case .timeout:
            return "Request timed out"
        case .offline:
            return "No network connection"
        case .duplicateUser(let email):
            return "User already exists: \(email)"
        case .forbidden(let reason):
            return "Operation not permitted: \(reason)"
        case .accountLocked:
            return "Account is locked"
        }
    }
}
```

According to the Swift API Design Guidelines, errors should be informative and actionable!
